22 ?- edge(a,b) = edge(C,b).
C = a.
% The terms can be unified because C can be instantiated to a, while b matches b.


23 ?- edge(a,d) = edge(C,B).
C = a,
B = d.
% The terms can be unified because C can be instantiated to a and B to d.


24 ?- edge(C,B) = edge(a,d).
C = a,
B = d.
% The terms can be unified because C and B can be instantiated to a and d.


25 ?- edge(A,f(g,h)) = edge(a,B).
A = a,
B = f(g, h).
% The terms can be unified because A can be instantiated to a and B to the structure f(g,h).



26 ?- edge(A,A) = edge(b,c).
false.
% The terms cannot be unified because the same variable A cannot be both b and c.


27 ?- edge(A,A) = edge(b,b).
A = b.
% The terms can be unified because both occurrences of A can be instantiated to b.


28 ?- A = B, B = f.
A = B, B = f.
% The terms can be unified because A and B are linked, and B is instantiated to f.

29 ?- A = B, B = f, A = e.
false.
